CREATE DATABASE RECIPEMATCHPLUS;
USE RECIPEMATCHPLUS;

CREATE TABLE USERS (
    users_id       INT           NOT NULL,
    username      VARCHAR(50)   NOT NULL,
    email         VARCHAR(100)  NOT NULL,
    passwords      VARCHAR(100)  NOT NULL,
    created_at    DATETIME      DEFAULT CURRENT_TIMESTAMP,
    modified_at   DATETIME      DEFAULT CURRENT_TIMESTAMP,
    PRIMARY KEY(users_id),
    UNIQUE(username),
    UNIQUE(email)
);

CREATE TABLE RECIPE (
    recipe_id        INT               NOT NULL,
    recipe_name      VARCHAR(100)      NOT NULL,
    cook_time        INT               NOT NULL,
    taste_type       VARCHAR(10)       NOT NULL,
    difficulty_level VARCHAR(10)       NOT NULL,
    added_by         INT               NOT NULL,
    created_at       DATETIME          DEFAULT CURRENT_TIMESTAMP,
    modified_at      DATETIME          DEFAULT CURRENT_TIMESTAMP,

    CHECK (taste_type IN('sweet', 'salty', 'sour', 'spicy')),
	CHECK (difficulty_level IN('easy', 'medium', 'hard')),
	PRIMARY KEY (recipe_id),
    FOREIGN KEY (added_by) REFERENCES USERS (users_id)
       ON DELETE CASCADE
       ON UPDATE CASCADE
);

CREATE TABLE INGREDIENT_CATEGORY (
    category_id    INT              NOT NULL,
    category_name  VARCHAR(100)     NOT NULL,
    parent_id      INT              NULL,
    created_at     DATETIME         DEFAULT CURRENT_TIMESTAMP,
    modified_at    DATETIME         DEFAULT CURRENT_TIMESTAMP,
    PRIMARY KEY (category_id),
    UNIQUE (category_name),
    FOREIGN KEY (parent_id) REFERENCES INGREDIENT_CATEGORY(category_id)
        ON DELETE NO ACTION
        ON UPDATE NO ACTION
);

CREATE TABLE INGREDIENT (
    ingredient_id    INT             NOT NULL,
    ingredient_name  VARCHAR(100)    NOT NULL,
    category_id      INT             NOT NULL,
    created_at       DATETIME        DEFAULT CURRENT_TIMESTAMP,
    modified_at      DATETIME        DEFAULT CURRENT_TIMESTAMP,

    PRIMARY KEY (ingredient_id),
    UNIQUE (ingredient_name),
    FOREIGN KEY (category_id) REFERENCES INGREDIENT_CATEGORY(category_id)
        ON DELETE CASCADE
        ON UPDATE CASCADE
);

CREATE TABLE PANTRY (
    users_id     INT              NOT NULL,
    pantry_no    INT              NOT NULL,
    created_at   DATETIME         DEFAULT CURRENT_TIMESTAMP,
    modified_at  DATETIME         DEFAULT CURRENT_TIMESTAMP,

    PRIMARY KEY (users_id, pantry_no),
    FOREIGN KEY (users_id) REFERENCES USERS(users_id)
       ON DELETE CASCADE
       ON UPDATE CASCADE
);

CREATE TABLE RECIPE_STEP (
    recipe_id     INT              NOT NULL,
    step_number   INT              NOT NULL,
    step_text     VARCHAR(1000)    NOT NULL,
    created_at    DATETIME         DEFAULT CURRENT_TIMESTAMP,
    modified_at   DATETIME         DEFAULT CURRENT_TIMESTAMP,

    PRIMARY KEY (recipe_id, step_number),
    FOREIGN KEY (recipe_id) REFERENCES RECIPE(recipe_id)
        ON DELETE CASCADE
        ON UPDATE CASCADE
);

CREATE TABLE STORED_IN (
    users_id       INT              NOT NULL,
    pantry_no      INT              NOT NULL,
    ingredient_id  INT              NOT NULL,
    quantity       DECIMAL(10,2),
    unit           VARCHAR(20),

    PRIMARY KEY (users_id, pantry_no, ingredient_id),    
    FOREIGN KEY (users_id, pantry_no) REFERENCES PANTRY(users_id, pantry_no)
        ON DELETE CASCADE
        ON UPDATE CASCADE,
    FOREIGN KEY (ingredient_id) REFERENCES INGREDIENT(ingredient_id)
        ON DELETE CASCADE
        ON UPDATE CASCADE
);

CREATE TABLE RECIPE_INGREDIENT (
    recipe_id      INT               NOT NULL,
    ingredient_id  INT               NOT NULL,
    amount_grams   DECIMAL(10,2),

    PRIMARY KEY (recipe_id, ingredient_id),
    FOREIGN KEY (recipe_id) REFERENCES RECIPE(recipe_id)
        ON DELETE CASCADE
        ON UPDATE CASCADE,
    FOREIGN KEY (ingredient_id) REFERENCES INGREDIENT(ingredient_id)
        ON DELETE CASCADE
        ON UPDATE CASCADE
);

CREATE TABLE BOOKMARK (
    users_id    INT               NOT NULL,
    recipe_id   INT               NOT NULL,
    created_at  DATETIME          DEFAULT CURRENT_TIMESTAMP,

    PRIMARY KEY (users_id, recipe_id),
    FOREIGN KEY (users_id) REFERENCES USERS(users_id)
        ON DELETE NO ACTION
        ON UPDATE NO ACTION,
    FOREIGN KEY (recipe_id) REFERENCES RECIPE(recipe_id)
        ON DELETE NO ACTION
        ON UPDATE NO ACTION
);

CREATE TABLE RATING (
    users_id    INT               NOT NULL,
    recipe_id   INT               NOT NULL,
    stars       INT               NOT NULL,
    created_at  DATETIME          DEFAULT CURRENT_TIMESTAMP,
    modified_at DATETIME          DEFAULT CURRENT_TIMESTAMP,

    PRIMARY KEY (users_id, recipe_id),
    CHECK (stars BETWEEN 1 AND 5),
    FOREIGN KEY (users_id) REFERENCES USERS(users_id)
        ON DELETE NO ACTION
        ON UPDATE NO ACTION,
    FOREIGN KEY (recipe_id) REFERENCES RECIPE(recipe_id)
        ON DELETE NO ACTION
        ON UPDATE NO ACTION
);

CREATE VIEW USER_RECIPE_MATCH AS
SELECT u.users_id, r.recipe_id, r.recipe_name,
CAST( ROUND ( ( SUM(CASE
    WHEN si.ingredient_id=r_ing.ingredient_id THEN 1
    WHEN cat_pantry.category_id=cat_recipe.category_id THEN 0.5
    ELSE 0 
    END)/COUNT(r_ing.ingredient_id))*100,0) AS INT) AS match_score
FROM USERS u
CROSS JOIN  RECIPE r
JOIN RECIPE_INGREDIENT r_ing ON r_ing.recipe_id=r.recipe_id
LEFT JOIN PANTRY p ON p.users_id=u.users_id
LEFT JOIN STORED_IN si ON si.users_id=p.users_id AND si.pantry_no=p.pantry_no
LEFT JOIN INGREDIENT ing_pantry ON si.ingredient_id=ing_pantry.ingredient_id
LEFT JOIN INGREDIENT_CATEGORY cat_pantry ON ing_pantry.category_id=cat_pantry.category_id
LEFT JOIN INGREDIENT ing_recipe ON r_ing.ingredient_id=ing_recipe.ingredient_id
LEFT JOIN INGREDIENT_CATEGORY cat_recipe ON ing_recipe.category_id=cat_recipe.category_id
GROUP BY u.users_id, r.recipe_id, r.recipe_name;

SELECT *
FROM USER_RECIPE_MATCH
ORDER BY users_id ASC, match_score DESC;

CREATE VIEW recipe_rating_summary AS
SELECT 
    r.recipe_id,
	r.recipe_name,
    CAST(AVG(rt.stars) AS DECIMAL (10, 2)) AS avg_rating,
    COUNT(rt.users_id) AS total_values
FROM RECIPE r
LEFT JOIN RATING rt 
	ON r.recipe_id = rt.recipe_id
GROUP BY r.recipe_id, r.recipe_name;

SELECT *
FROM recipe_rating_summary

CREATE TRIGGER update_recipe_modified_at ON RECIPE
AFTER UPDATE
AS
	UPDATE RECIPE
	SET modified_at = CURRENT_TIMESTAMP
	WHERE recipe_id IN (SELECT recipe_id FROM inserted);

CREATE TRIGGER update_recipe_modified_on_rating ON RATING
AFTER INSERT
AS
    UPDATE RECIPE
    SET modified_at = CURRENT_TIMESTAMP
    WHERE recipe_id IN (SELECT recipe_id FROM inserted);

CREATE TRIGGER prevent_negative_stock ON STORED_IN
AFTER UPDATE
AS
	UPDATE s
	SET s.quantity=0
	FROM STORED_IN s
	JOIN inserted i ON s.users_id=i.users_id AND s.pantry_no=i.pantry_no AND s.ingredient_id=i.ingredient_id
	WHERE i.quantity<0;
